function toNumber(value) {
    const number = Number(value);
    return Number.isFinite(number) ? number : 0;
}

function formatNumber(value) {
    return new Intl.NumberFormat("en-IN").format(toNumber(value));
}

function normalizeAnalytics(response) {
    // Supports both:
    // response
    // response.data
    // response.data.data
    const root =
        response?.data?.data ??
        response?.data ??
        response ??
        {};

    return {
        generatedAt: root.generated_at ?? null,

        hospital: {
            patients: toNumber(root.hospital?.patients),
            doctors: toNumber(root.hospital?.doctors),
            nurses: toNumber(root.hospital?.nurses),
            staff: toNumber(root.hospital?.staff),
            departments: toNumber(root.hospital?.departments),
            wards: toNumber(root.hospital?.wards),
        },

        capacity: {
            totalBeds: toNumber(root.capacity?.total_beds),
            occupiedBeds: toNumber(root.capacity?.occupied_beds),
            availableBeds: toNumber(root.capacity?.available_beds),
            reservedBeds: toNumber(root.capacity?.reserved_beds),
            maintenanceBeds: toNumber(root.capacity?.maintenance_beds),
            occupancy: toNumber(root.capacity?.occupancy_percentage),
        },

        appointments: {
            total: toNumber(root.appointments?.total),
            today: toNumber(root.appointments?.today),
            scheduled: toNumber(root.appointments?.scheduled),
            confirmed: toNumber(root.appointments?.confirmed),
            completed: toNumber(root.appointments?.completed),
            cancelled: toNumber(root.appointments?.cancelled),
            completion: toNumber(
                root.appointments?.completion_percentage
            ),
        },

        admissions: {
            total: toNumber(root.patient_flow?.total_admissions),
            active: toNumber(root.patient_flow?.active_admissions),
            discharged: toNumber(
                root.patient_flow?.discharged_admissions
            ),
        },

        emergency: {
            total: toNumber(root.emergency?.total_cases),
            active: toNumber(root.emergency?.active_cases),
            today: toNumber(root.emergency?.today_cases),
            critical: toNumber(root.emergency?.critical_triage),
            emergency: toNumber(root.emergency?.emergency_triage),
            urgent: toNumber(root.emergency?.urgent_triage),
        },

        surgical: {
            total: toNumber(
                root.surgical_services?.total_surgeries
            ),
            scheduled: toNumber(
                root.surgical_services?.scheduled
            ),
            completed: toNumber(
                root.surgical_services?.completed
            ),
            theatres: toNumber(
                root.surgical_services?.operation_theatres
            ),
            activeTheatres: toNumber(
                root.surgical_services?.active_theatres
            ),
        },

        clinical: {
            prescriptions: toNumber(
                root.clinical_services?.prescriptions
            ),
            todayPrescriptions: toNumber(
                root.clinical_services?.today_prescriptions
            ),
            laboratoryTests: toNumber(
                root.clinical_services?.laboratory_tests
            ),
            radiologyRecords: toNumber(
                root.clinical_services?.radiology_records
            ),
        },

        pharmacy: {
            pharmacies: toNumber(
                root.pharmacy_inventory?.pharmacies
            ),
            inventoryItems: toNumber(
                root.pharmacy_inventory?.inventory_items
            ),
            lowStock: toNumber(
                root.pharmacy_inventory?.low_stock_items
            ),
        },

        financial: {
            invoiceTotal: toNumber(
                root.financial?.invoice_total
            ),
            invoicePaid: toNumber(
                root.financial?.invoice_paid
            ),
            invoiceBalance: toNumber(
                root.financial?.invoice_balance
            ),
            paymentCollection: toNumber(
                root.financial?.payment_collection
            ),
        },

        alerts: Array.isArray(root.alerts)
            ? root.alerts
            : [],
    };
}

function StatCard({
    icon: Icon,
    label,
    value,
    detail,
    tone = "cyan",
}) {
    const tones = {
        cyan: "border-cyan-400/10 bg-cyan-400/[0.04] text-cyan-300",
        blue: "border-blue-400/10 bg-blue-400/[0.04] text-blue-300",
        violet:
            "border-violet-400/10 bg-violet-400/[0.04] text-violet-300",
        emerald:
            "border-emerald-400/10 bg-emerald-400/[0.04] text-emerald-300",
        amber:
            "border-amber-400/10 bg-amber-400/[0.04] text-amber-300",
        rose: "border-rose-400/10 bg-rose-400/[0.04] text-rose-300",
    };

    return (
        <div className="group rounded-2xl border border-white/[0.07] bg-[#09111f]/80 p-5 transition duration-300 hover:-translate-y-0.5 hover:border-white/[0.12]">
            <div className="flex items-start justify-between gap-4">
                <div>
                    <p className="text-[11px] font-semibold uppercase tracking-[0.18em] text-slate-500">
                        {label}
                    </p>

                    <p className="mt-3 text-3xl font-bold tracking-tight text-white">
                        {formatNumber(value)}
                    </p>

                    <p className="mt-2 text-xs text-slate-500">
                        {detail}
                    </p>
                </div>

                <div
                    className={`rounded-xl border p-3 ${tones[tone] || tones.cyan}`}
                >
                    <Icon size={21} />
                </div>
            </div>
        </div>
    );
}

function SectionHeader({ icon: Icon, title, subtitle }) {
    return (
        <div className="mb-5 flex items-center gap-3">
            <div className="rounded-xl border border-cyan-400/10 bg-cyan-400/[0.05] p-2.5 text-cyan-300">
                <Icon size={18} />
            </div>

            <div>
                <h2 className="text-sm font-semibold text-white">
                    {title}
                </h2>
                <p className="text-xs text-slate-500">
                    {subtitle}
                </p>
            </div>
        </div>
    );
}

                                                <p className="mt-1 text-xs leading-5 text-slate-500">
                                                    {
                                                        alert.message
                                                    }
                                                </p>
                                            </div>
                                        </div>
                                    </div>
                                )
                            )}
                        </div>
                    )}
                </div>

                {/* Footer telemetry */}
                <div className="flex flex-col justify-between gap-2 border-t border-white/[0.06] pt-4 text-[11px] text-slate-600 md:flex-row">
                    <span>
                        Live hospital intelligence • PostgreSQL
                        backed
                    </span>

                    <span className="flex items-center gap-2">
                        <Clock3 size={13} />
                        Last backend snapshot:{" "}
                        {data.generatedAt
                            ? new Date(
                                  data.generatedAt
                              ).toLocaleTimeString(
                                  "en-IN"
                              )
                            : "Unavailable"}
                    </span>
                </div>
            </div>
        </div>
    );
}

function MiniMetric({ label, value }) {
    return (
        <div className="rounded-xl border border-white/[0.06] bg-white/[0.02] p-3">
            <p className="text-[10px] font-semibold uppercase tracking-[0.12em] text-slate-600">
                {label}
            </p>
            <p className="mt-2 text-lg font-bold text-slate-200">
                {formatNumber(value)}
            </p>
        </div>
    );
}
